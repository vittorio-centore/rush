Export the email function (per D-09, D-12):

```typescript
export async function sendApplicationConfirmation(
  to: string,
  clubName: string,
  submittedAt: string,
): Promise<{ error: Error | null }> {
  const apiKey = process.env.RESEND_API_KEY;

  if (!apiKey) {
    return { error: new Error("Missing RESEND_API_KEY") };
  }

  const resend = new Resend(apiKey);
  const { error } = await resend.emails.send({
    from: "Rush <noreply@rush.app>",
    to,
    subject: `Your application to ${clubName} has been received`,
    html: buildConfirmationEmailHtml(clubName, submittedAt),
  });

  return { error: error as Error | null };
}
```

Note the `from` uses `noreply@rush.app` per D-12, distinct from the deadline reminder's `reminders@rush.app`.

```typescript
// Build a helper to get the visible stage label per D-06
const appStageIds = apps
  .map((a) => a.stage_id)
  .filter((id): id is string => id !== null);

const uniqueStageIds = [...new Set(appStageIds)];
const { data: stageData } = uniqueStageIds.length > 0
  ? await supabase
      .from("club_pipeline_stages")
      .select("id, label, is_released")
      .in("id", uniqueStageIds)
  : { data: [] };

const stageMap = new Map(
  (stageData ?? []).map((s: { id: string; label: string; is_released: boolean }) => [s.id, s]),
);

// For apps whose current stage is NOT released, we need to walk transitions
// to find the last released stage they passed through.
const appsNeedingFallback = apps.filter((a) => {
  if (!a.stage_id) return false;
  const stage = stageMap.get(a.stage_id);
  return !stage?.is_released;
});

const fallbackLabels = new Map<string, string>();

if (appsNeedingFallback.length > 0) {
  const fallbackAppIds = appsNeedingFallback.map((a) => a.id);
  const { data: transitionsData } = await supabase
    .from("club_application_stage_transitions")
    .select("application_id, to_stage_id, changed_at")
    .in("application_id", fallbackAppIds)
    .order("changed_at", { ascending: false });

  // For each app, find the most recent transition whose to_stage is released
  // We need to fetch all the to_stage_ids that appear in transitions
  const transitionStageIds = [...new Set(
    (transitionsData ?? [])
      .map((t: { to_stage_id: string | null }) => t.to_stage_id)
      .filter((id): id is string => id !== null)
  )];

  const { data: transitionStages } = transitionStageIds.length > 0
    ? await supabase
        .from("club_pipeline_stages")
        .select("id, label, is_released")
        .in("id", transitionStageIds)
    : { data: [] };

  const transitionStageMap = new Map(
    (transitionStages ?? []).map((s: { id: string; label: string; is_released: boolean }) => [s.id, s]),
  );

  for (const app of appsNeedingFallback) {
    const appTransitions = (transitionsData ?? []).filter(
      (t: { application_id: string }) => t.application_id === app.id,
    );
    for (const t of appTransitions) {
      const stage = t.to_stage_id ? transitionStageMap.get(t.to_stage_id) : null;
      if (stage?.is_released) {
        fallbackLabels.set(app.id, stage.label);
        break;
      }
    }
  }
}

// Helper to get the visible stage label for a given app
function getVisibleStageLabel(app: Application): string | null {
  if (app.stage_id) {
    const stage = stageMap.get(app.stage_id);
    if (stage?.is_released) return stage.label;
  }
  return fallbackLabels.get(app.id) ?? null;
}
```
